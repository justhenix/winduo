cmake_minimum_required(VERSION 3.24)
project(WinDuo LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "WinDuo requires MSVC and the Windows SDK")
endif()
add_executable(WinDuo WIN32 native/app.cpp)
target_compile_features(WinDuo PRIVATE cxx_std_20)
set_property(TARGET WinDuo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(WinDuo PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(WinDuo PRIVATE /W4 /utf-8 /EHsc)
target_link_options(WinDuo PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest" /DYNAMICBASE /NXCOMPAT /DELAYLOAD:MF.dll /DELAYLOAD:MFPlat.dll /DELAYLOAD:MFReadWrite.dll)
target_link_libraries(WinDuo PRIVATE user32 gdi32 shell32 advapi32 ole32 runtimeobject windowsapp wtsapi32 powrprof comctl32 d3d11 dxgi bcrypt d2d1 mfplat mfreadwrite mfuuid mf delayimp dxguid dwmapi uxtheme)
set_target_properties(WinDuo PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/out")
